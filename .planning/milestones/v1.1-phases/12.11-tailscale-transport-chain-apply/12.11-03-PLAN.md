```typescript
// Build panel credentials map using Tailscale transport resolution
const panelCredentials = new Map<number, { panelUrl: string; apiKey: string }>();

// Fetch all active remote panels
const remotePanels = await prisma.remotePanel.findMany({
  where: { isActive: true },
});

// For each chain node, find matching RemotePanel + Server and resolve transport
for (const node of chainConfig.nodes) {
  // Try to find a RemotePanel whose panelUrl hostname matches this node's server
  // (RemotePanel has no serverId FK, so we match by hostname/correlation)
  let matchedPanel = remotePanels.find((p) =>
    p.panelUrl.includes(node.hostname) || node.hostname.includes(new URL(p.panelUrl).hostname)
  );

  // If no hostname match, try matching by serverId via the serverMapping
  if (!matchedPanel) {
    const mappedServerId = node.serverId;
    matchedPanel = remotePanels.find((p) => p.id === mappedServerId);
  }

  if (!matchedPanel) {
    // No matching panel -- this node will be skipped with error by applyChainConfig
    continue;
  }

  // Get API key from caller-provided panelApiKeys, keyed by panelId
  const apiKey = panelApiKeys?.[matchedPanel.id] ?? panelApiKeys?.[node.serverId];
  if (!apiKey) {
    // No API key provided -- this node will be skipped with error by applyChainConfig
    continue;
  }

  // Resolve Tailscale transport address for the server behind this panel
  const server = await prisma.server.findFirst({
    where: { id: node.serverId },
    select: { id: true, tailnetIP: true, tailnetHostname: true, hostname: true },
  });

  let panelUrl = matchedPanel.panelUrl; // Default to registered panelUrl

  if (server) {
    const { resolvePanelTransport } = await import('@/lib/transport-resolver');
    const transport = await resolvePanelTransport(server, { panelUrl: matchedPanel.panelUrl });
    if (transport) {
      panelUrl = transport.panelUrl;
    }
  }

  panelCredentials.set(node.serverId, { panelUrl, apiKey });
}
```
