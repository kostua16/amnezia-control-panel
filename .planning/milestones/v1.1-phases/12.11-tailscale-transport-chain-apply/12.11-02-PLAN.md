```typescript
// Resolve transport for each server in the chain
const { prisma } = await import('./prisma');

const resolvedNodes: Array<ChainNode & { hostname: string; port: number }> =
  template.nodes.map(async (node, index) => {
    const serverId = serverMapping[index];
    if (serverId === undefined) {
      throw new Error(`No server mapping for node index ${index}`);
    }
    const server = serverLookup.get(serverId);
    if (!server) {
      throw new Error(`Server ${serverId} not found for node "${node.label}"`);
    }

    // Resolve WireGuard service port from Service model
    // WireGuard typically uses port 51820; look up actual service port if available
    const wireguardService = await prisma.service.findFirst({
      where: { serverId: server.id, type: 'AWG' },
      select: { port: true },
    });
    const wireguardPort = wireguardService?.port ?? 51820;

    // Resolve Tailscale transport address
    const transport = await resolvePanelTransport(
      server,
      { panelUrl: `https://${server.tailnetIP ?? server.hostname}:${wireguardPort}` },
      wireguardPort,
    );

    if (transport) {
      return {
        ...node,
        serverId: server.id,
        hostname: transport.tailscaleIP,
        port: wireguardPort,
      };
    }

    // Fallback: use raw server data (with warning)
    console.warn(`[chain-router] Transport resolution failed for server ${server.id} (${server.hostname}), falling back to raw hostname:port`);
    return {
      ...node,
      serverId: server.id,
      hostname: server.hostname,
      port: wireguardPort,
    };
  });

// Await all resolution promises
const nodes = await Promise.all(resolvedNodes);
```

```typescript
// Resolve template nodes with panel data and Tailscale transport
const resolvedNodes = await Promise.all(
  template.nodes.map(async (node, index) => {
    const panelId = panelMapping[index];
    const panel = panelLookup.get(panelId)!;

    // Resolve WireGuard service port from Service model
    const wireguardService = await prisma.service.findFirst({
      where: { serverId: panel.id, type: 'AWG' },
      select: { port: true },
    });
    const wireguardPort = wireguardService?.port ?? 51820;

    // Try Tailscale transport resolution
    const server = await prisma.server.findFirst({
      where: {
        OR: [
          { tailnetIP: { not: null } },
          { tailnetHostname: { not: null } },
          { hostname: { not: null } },
        ],
      },
      select: { id: true, tailnetIP: true, tailnetHostname: true, hostname: true },
    });

    const transport = server
      ? await resolvePanelTransport(server, { panelUrl: panel.panelUrl }, wireguardPort)
      : null;

    if (transport) {
      return {
        ...node,
        serverId: panel.id,
        hostname: transport.tailscaleIP,
        port: wireguardPort,
      };
    }

    // Fallback: parse from panelUrl (legacy behavior)
    console.warn(`[chain-config] Transport resolution failed for panel ${panel.name}, falling back to panelUrl parsing`);
    let hostPart = panel.panelUrl.replace(/^https?:\/\//, '').split('/')[0];
    let port = wireguardPort;
    if (hostPart.includes(':')) {
      const [host, portStr] = hostPart.split(':');
      hostPart = host;
      port = parseInt(portStr, 10) || wireguardPort;
    }
    return {
      ...node,
      serverId: panel.id,
      hostname: hostPart,
      port,
    };
  }),
);
```
