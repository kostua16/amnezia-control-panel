```typescript
/** A single line in a config diff */
export interface ConfigDiffLine {
  type: 'added' | 'removed' | 'unchanged';
  content: string;
}

/** A section of the diff (e.g., "WireGuard Peers", "Routing Rules") */
export interface ConfigDiffSection {
  label: string;
  lines: ConfigDiffLine[];
  /** Summary of changes in this section */
  summary: { added: number; removed: number; unchanged: number };
}

/** Diff result for a single panel */
export interface ConfigDiffResult {
  panelId: number;
  panelName: string;
  hasChanges: boolean;
  sections: ConfigDiffSection[];
  /** The full current config as formatted JSON string (for display) */
  currentConfigFormatted: string | null;
  /** The full new config as formatted JSON string (for display) */
  newConfigFormatted: string;
}
```

Implement **computeConfigDiff(panelId: number, panelName: string, newConfig: PanelSyncPayload): Promise<ConfigDiffResult>**:

1. Fetch current cached config from DB: `prisma.cachedPanelConfig.findUnique({ where: { panelId } })`
2. Parse current config from `cachedConfig.config` as PanelSyncPayload (or null if none exists)
3. Build diff sections by comparing current and new config:

   **Section 1: "Chain Nodes"** (`chainNodes`)
   - Compare by label (unique identifier per node)
   - For each node in new config: if not in current -> added, if in current but different fields -> removed + added, if identical -> unchanged
   - For each node in current but not in new -> removed
   - Format: `"{role} {protocol}: {label} ({hostname}:{port})"`

   **Section 2: "WireGuard Peers"** (`wireguardPeers`)
   - Compare by publicKey (unique identifier per peer)
   - Format: `"Peer {publicKey.substring(0, 12)}... -> {endpoint} [{allowedIPs}]"`

   **Section 3: "Routing Rules"** (`routingRules`)
   - Compare by composite key: `type + value + outboundTag + priority`
   - Format: `"{type}/{value} -> {outboundTag} (priority {priority})"`

4. Each section produces ConfigDiffLine[] with type 'added', 'removed', or 'unchanged'
5. Compute summary counts per section
6. Set hasChanges = true if any section has added or removed lines
7. Set currentConfigFormatted = `JSON.stringify(currentConfig, null, 2)` or null
8. Set newConfigFormatted = `JSON.stringify(newConfig, null, 2)`

Implement **formatDiffForDisplay(sections: ConfigDiffSection[]): string**:
- Returns a human-readable diff string for logging/debugging
- Format: section headers with lines indented, prefixed with `+`, `-`, or ` `
- Example:
  ```
  WireGuard Peers (+2, -1)
  + Peer abc123... -> 10.0.0.2:51820 [0.0.0.0/0]
  - Peer def456... -> 10.0.0.3:51820 [10.0.0.1/32]
    Peer ghi789... -> 10.0.0.1:51820 [0.0.0.0/0]
  ```
